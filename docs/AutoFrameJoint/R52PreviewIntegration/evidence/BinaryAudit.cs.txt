using System;
using System.IO;
using System.Linq;
using System.Text;
using System.Collections.Generic;
using System.Security.Cryptography;
using System.Web.Script.Serialization;
using Mono.Cecil;
using Mono.Cecil.Cil;
class BinaryAudit {
 static string Hash(string s){using(var h=SHA256.Create())return BitConverter.ToString(h.ComputeHash(Encoding.UTF8.GetBytes(s))).Replace("-","").ToLowerInvariant();}
 static string IL(MethodDefinition m){if(!m.HasBody)return "NO_BODY";var ins=m.Body.Instructions.Where(i=>i.OpCode.Code!=Code.Nop).ToList();Func<Instruction,int> index=i=>{int n=m.Body.Instructions.IndexOf(i);while(n<m.Body.Instructions.Count&&m.Body.Instructions[n].OpCode.Code==Code.Nop)n++;return n>=m.Body.Instructions.Count?ins.Count:ins.IndexOf(m.Body.Instructions[n]);};
 return String.Join("\n",ins.Select(i=>{string op=i.OpCode.Name.EndsWith(".s")?i.OpCode.Name.Substring(0,i.OpCode.Name.Length-2):i.OpCode.Name;object v=i.Operand;string value=v is Instruction?"target:"+index((Instruction)v):v is Instruction[]?String.Join(",",((Instruction[])v).Select(x=>index(x))):v is MethodReference?((MethodReference)v).FullName:v is FieldReference?((FieldReference)v).FullName:v is TypeReference?((TypeReference)v).FullName:v is VariableDefinition?"var:"+((VariableDefinition)v).Index:v is ParameterDefinition?"arg:"+((ParameterDefinition)v).Index:v==null?"":Convert.ToString(v,System.Globalization.CultureInfo.InvariantCulture);return op+" "+value;}));}
 static object Record(MethodDefinition m,Dictionary<string,MethodDefinition> prior){string text=IL(m);MethodDefinition b;return new{method=m.FullName,normalized_il_sha256=Hash(text),baseline_method_found=prior.TryGetValue(m.FullName,out b),matches_r51_harness_il=b!=null&&IL(b)==text,calls=m.HasBody?m.Body.Instructions.Where(i=>i.Operand is MethodReference).Select(i=>((MethodReference)i.Operand).FullName).Distinct().ToArray():new string[0],strings=m.HasBody?m.Body.Instructions.Where(i=>i.OpCode.Code==Code.Ldstr).Select(i=>(string)i.Operand).ToArray():new string[0]};}
 static int Main(string[] args){using(var current=ModuleDefinition.ReadModule(args[0],new ReaderParameters{ReadSymbols=false,InMemory=true}))using(var baseline=ModuleDefinition.ReadModule(args[1],new ReaderParameters{ReadSymbols=false,InMemory=true})){var types=current.GetTypes().Where(t=>t.FullName.StartsWith("ADDIN.Commands.AutoFrameJoint.")||t.FullName.StartsWith("ADDIN.UI.AutoFrameJoint.")).ToArray();var prior=baseline.GetTypes().SelectMany(t=>t.Methods).ToDictionary(m=>m.FullName);var rows=types.Select(t=>new{type=t.FullName,methods=t.Methods.Select(m=>Record(m,prior)).ToArray()}).ToArray();var report=new{file=Path.GetFileName(args[0]),mvid=current.Mvid,expected_log_mvid="2f4e359e-bffe-4f74-9441-891bb4c77b12",matches_user_log=current.Mvid==new Guid("2f4e359e-bffe-4f74-9441-891bb4c77b12"),baseline_file=Path.GetFileName(args[1]),baseline_mvid=baseline.Mvid,types=rows};string json=new JavaScriptSerializer{MaxJsonLength=Int32.MaxValue,RecursionLimit=200}.Serialize(report);File.WriteAllText(args[2],json);Console.WriteLine("MVID="+current.Mvid+"; matches user log="+report.matches_user_log+"; AFJ types="+types.Length);var preview=types.Single(t=>t.Name=="AutoFrameJointGhostPreview").Methods.Single(m=>m.Name=="TryShowCutCandidate");Console.WriteLine("R51 marker="+preview.Body.Instructions.Any(i=>i.Operand is string&&((string)i.Operand).Contains("[AFJ-R51]"))+"; solver call="+preview.Body.Instructions.Any(i=>i.Operand is MethodReference&&((MethodReference)i.Operand).DeclaringType.Name=="AfjJointSolver"));}return 0;}
}
