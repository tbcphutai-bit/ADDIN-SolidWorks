 static object StockBeforeCuts(ISldWorks app,ModelDoc2 model,Component2 a,Component2 b)
 {
  var rows=new List<object>();
  foreach(var c in new[]{a,b})
  {
   var part=(ModelDoc2)c.GetModelDoc2();int activation=0;
   app.ActivateDoc3(part.GetTitle(),false,(int)swRebuildOnActivation_e.swDontRebuildActiveDoc,ref activation);
   Assert(app.IsSame(app.ActiveDoc,part)==(int)swObjectEquality.swObjectSame,"Cannot activate copied stock Part");
   if(part.ConfigurationManager.ActiveConfiguration.Name!=c.ReferencedConfiguration)Assert(part.ShowConfiguration2(c.ReferencedConfiguration),"Cannot activate folded copy configuration");
   var cuts=new List<Feature>();
   for(Feature f=part.FirstFeature() as Feature;f!=null;f=f.GetNextFeature() as Feature)
    if(f.GetTypeName2()=="Cut"||(f.GetTypeName2()=="ICE"&&f.GetTypeName()=="Cut"))cuts.Add(f);
   foreach(Feature f in cuts)
   {bool ok=f.SetSuppression2((int)swFeatureSuppressionAction_e.swSuppressFeature,(int)swInConfigurationOpts_e.swThisConfiguration,null);Assert(ok&&f.IsSuppressed(),"Copy-only stock setup could not suppress "+f.Name);}
   rows.Add(D("component",c.Name2,"state","ALL_EXISTING_CUTS_SUPPRESSED_ON_DISPOSABLE_COPY","feature_names",cuts.Select(f=>f.Name).ToArray()));
  }
  int error=0;app.ActivateDoc3(model.GetTitle(),false,(int)swRebuildOnActivation_e.swDontRebuildActiveDoc,ref error);
  Assert(app.IsSame(app.ActiveDoc,model)==(int)swObjectEquality.swObjectSame,"Copied assembly not active after stock setup");
  return rows;
 }
 static object Case(ISldWorks app,string input,string root,string name,string suffixA,string suffixB)
 {

  string assemblyPath=Directory.GetFiles(folder,"*.SLDASM").Single();

   foreach(var component in components)
   {

   }
   Component2 a=components.Single(c=>c.Name2.EndsWith(suffixA,StringComparison.OrdinalIgnoreCase));
   Component2 b=components.Single(c=>c.Name2.EndsWith(suffixB,StringComparison.OrdinalIgnoreCase));
   if(System.Environment.GetEnvironmentVariable("AFJ_TEST_STOCK_BEFORE_CUTS")=="1")row["stock_setup"]=StockBeforeCuts(app,model,a,b);
   if(System.Environment.GetEnvironmentVariable("AFJ_TEST_SUPPRESS_OTHER_OCCURRENCES")=="1")
   {
    var suppressed=new List<object>();
    foreach(var c in components)
    {
     if(app.IsSame(c,a)==(int)swObjectEquality.swObjectSame||app.IsSame(c,b)==(int)swObjectEquality.swObjectSame)continue;
     if(!AfjNativeCutCommit.SameReferencePath(c.GetPathName(),a.GetPathName())&&!AfjNativeCutCommit.SameReferencePath(c.GetPathName(),b.GetPathName()))continue;
     int code=c.SetSuppression2((int)swComponentSuppressionState_e.swComponentSuppressed);
     Assert(c.IsSuppressed(),"Cannot suppress duplicate occurrence on test copy");
     suppressed.Add(D("component",c.Name2,"suppression_result",code,"model_document_null",c.GetModelDoc2()==null));
    }
    row["suppressed_other_occurrences"]=suppressed;
   }
   ModelDoc2 pa,pb;MathTransform ta,tb;
   Body2 beforeA=AfjJointSolver.NativeBody(a,out pa,out ta),beforeB=AfjJointSolver.NativeBody(b,out pb,out tb);
   parts.Add(pa);parts.Add(pb);

   foreach(var part in parts)Assert(Path.GetFullPath(part.GetPathName()).StartsWith(Path.GetFullPath(folder)+Path.DirectorySeparatorChar,StringComparison.OrdinalIgnoreCase),"Reference escaped new disposable copy folder");
   var solveTimer=Stopwatch.StartNew();Trace.WriteLine("[AFJ-HARNESS] SOLVE_BEGIN utc="+DateTime.UtcNow.ToString("o"));

   if(name=="rollback_after_both_parts")
   {
    // Deliberate mismatch at the final B validation; production's private ticket
    // is not exposed to this mutation. Exercise real native two-Part rollback.
    solution.B.Kept.Clear();solution.B.Kept.Add(AfjTemporaryBoolean.Copy(beforeB));
   }
   if(name=="stale_preview_rejected")ticket.IsSourceCurrent=()=>false;

    try
    {

   }
   var applyTimer=Stopwatch.StartNew();Trace.WriteLine("[AFJ-HARNESS] APPLY_BEGIN utc="+DateTime.UtcNow.ToString("o"));

   row["apply"]=report;row["returned_success"]=applied;row["message"]=message;
   if(name=="positive")
   {
    Assert(applied&&report.Cuts.Count>0,"Expected actual Cut-Extrude features");

    row["cross_frame_end_reference_undo_test"]=crossUndoRow;
    LinkEndsUndo(app,model,a,b,solution,report,crossUndoRow,Path.Combine(folder,"cross_end_undo.private.json"));
   }
   else
   {
    Assert(!applied&&report.RollbackVerified,"Expected rejection/verified rollback");
    Assert(AfjNativeCutCommit.SymmetricDifference(AfjTemporaryBoolean.Copy(AfjNativeCutCommit.CurrentBody(pa)),beforeA)<=1e-12,"A not restored");
    Assert(AfjNativeCutCommit.SymmetricDifference(AfjTemporaryBoolean.Copy(AfjNativeCutCommit.CurrentBody(pb)),beforeB)<=1e-12,"B not restored");
    if(name=="rollback_after_both_parts")Assert(report.Cuts.Count>=2&&report.Rollback.Count==2,"Both native Part histories not exercised");
    if(name=="shared_part_rejected")Assert(report.Cuts.Count==0&&report.Error.Contains("SHARED_PART"),"Expected shared-Part scope refusal before native Feature creation");
   }
   row["status"]="PASS_NATIVE_API";
  }
  catch(Exception ex)
  {
   row["status"]="FAIL";row["error"]=ex.ToString();row["hresult"]="0x"+ex.HResult.ToString("X8");
   if(name=="unsupported_current"&&ex.Message.StartsWith("UNVERIFIED_RULE",StringComparison.Ordinal)&&parts.Count==2)
   {bool unchanged=parts.Select((p,i)=>p.GetUpdateStamp()==stamps[i]&&Tree(p).SequenceEqual(trees[i])).All(v=>v);row["no_part_history_change"]=unchanged;row["native_writer_called"]=false;if(unchanged)row["status"]="PASS_NATIVE_REFUSAL";}
  }
  finally
  {
   try{row["copy_file_hashes_unchanged_before_close"]=sources.All(p=>Hash(Path.Combine(folder,Path.GetFileName(p)))==original[p]);}catch(Exception ex){row["preclose_hash_error"]=ex.ToString();}
   var ownedPaths=new List<string>();
   for(ModelDoc2 d=app.GetFirstDocument() as ModelDoc2;d!=null;d=d.GetNext() as ModelDoc2)
   {string path=d.GetPathName();if(!String.IsNullOrEmpty(path)&&Path.GetFullPath(path).StartsWith(Path.GetFullPath(folder)+Path.DirectorySeparatorChar,StringComparison.OrdinalIgnoreCase))ownedPaths.Add(path);}
   foreach(string path in ownedPaths.OrderBy(p=>Path.GetExtension(p).Equals(".SLDASM",StringComparison.OrdinalIgnoreCase)?0:1))

   File.WriteAllText(Path.Combine(folder,"result.private.json"),json.Serialize(row));
  }
  return row;
 }

 static Face2 MateFace(Component2 other,MathTransform ownTransform,AfjToolDefinition tool,double gap)
 {
  MathTransform ot=other.GetTotalTransform(false);double[] om=(double[])ot.ArrayData,tm=(double[])ownTransform.ArrayData;
  double[] worldPoint=AfjProfileAnalyzer.Transform(tool.Polygon[0],tm),n=AfjParametricRules.Unit(AfjParametricRules.Sub(AfjProfileAnalyzer.Transform(tool.Direction,tm),AfjProfileAnalyzer.Transform(new double[3],tm)));
  double value=AfjParametricRules.Dot(worldPoint,n)+gap;
  var candidates=new List<Face2>();
  foreach(Face2 face in (Array)AfjNativeCutCommit.CurrentBody((ModelDoc2)other.GetModelDoc2()).GetFaces()){
   var surface=(Surface)face.GetSurface();if(!surface.IsPlane())continue;var plane=(double[])surface.PlaneParams;
   var normal=AfjParametricRules.Unit(AfjParametricRules.Sub(AfjProfileAnalyzer.Transform(plane.Take(3).ToArray(),om),AfjProfileAnalyzer.Transform(new double[3],om)));
   var point=AfjProfileAnalyzer.Transform(plane.Skip(3).Take(3).ToArray(),om);
   var owner=face.GetFeature() as Feature;if(owner!=null&&owner.Name.StartsWith("AFJ_",StringComparison.Ordinal))continue;
   if(Math.Abs(AfjParametricRules.Dot(normal,n))>.999999&&Math.Abs(AfjParametricRules.Dot(point,n)-value)<1e-7)candidates.Add(face);
  }
  Assert(candidates.Count>0,"No original native mating support Face at planned cut+gap plane");return candidates.OrderByDescending(f=>f.GetArea()).First();
 }
 static void ValidatePair(ISldWorks app,ModelDoc2 assembly,Component2 a,Component2 b,AfjJointSolution solution,string phase,Dictionary<string,object> row)
 {
  var pa=(ModelDoc2)a.GetModelDoc2();var pb=(ModelDoc2)b.GetModelDoc2();
  double da=AfjNativeCutCommit.SymmetricDifference(AfjTemporaryBoolean.Copy(AfjNativeCutCommit.CurrentBody(pa)),solution.A.Kept.Single());
  double db=AfjNativeCutCommit.SymmetricDifference(AfjTemporaryBoolean.Copy(AfjNativeCutCommit.CurrentBody(pb)),solution.B.Kept.Single());
  row[phase]=D("symmetric_a_m3",da,"symmetric_b_m3",db);Assert(da<=1e-12&&db<=1e-12,"External end reference changed reviewed geometry at "+phase);
 }
 static List<object> FullTree(ModelDoc2 model)
 {
  var rows=new List<object>();int total=0;
  Action<Feature,bool,string,int> visit=null;
  visit=(start,top,parent,depth)=>{
   Assert(depth<16,"Full feature-tree depth limit reached");var siblings=new HashSet<int>();int index=0;
   for(Feature f=start;f!=null;f=(top?f.GetNextFeature():f.GetNextSubFeature()) as Feature){
    Assert(++total<3000,"Full feature-tree enumeration limit reached");int id=f.GetID();Assert(siblings.Add(id),"Full feature-tree enumeration repeated a sibling Feature ID");
    string path=parent+"/"+(index++).ToString(System.Globalization.CultureInfo.InvariantCulture)+":"+id.ToString(System.Globalization.CultureInfo.InvariantCulture);
    string type=f.GetTypeName2(),underlying=type=="ICE"?f.GetTypeName():type;
    rows.Add(D("path",path,"depth",depth,"id",id,"type",type,"underlying",underlying,"name",f.Name,"suppressed",f.IsSuppressed()));
    Feature sub=f.GetFirstSubFeature() as Feature;if(sub!=null)visit(sub,false,path,depth+1);
   }
  };
  visit(model.FirstFeature() as Feature,true,"",0);return rows;
 }
 static string CanonicalTree(List<object> tree){return String.Join("\n",tree.Cast<Dictionary<string,object>>().Select(r=>json.Serialize(new object[]{r["path"],r["depth"],r["id"],r["type"],r["underlying"],r["name"],r["suppressed"]})));}
 static int[] HolderIds(List<object> tree){return tree.Cast<Dictionary<string,object>>().Where(r=>((string)r["type"]).IndexOf("InContext",StringComparison.OrdinalIgnoreCase)>=0).Select(r=>(int)r["id"]).Distinct().OrderBy(x=>x).ToArray();}

 }
 static void LinkEndsUndo(ISldWorks app,ModelDoc2 model,Component2 a,Component2 b,AfjJointSolution solution,AfjNativeApplyReport report,Dictionary<string,object> row,string evidencePath)
 {
  var assy=(AssemblyDoc)model;var components=new[]{a,b};var parts=components.Select(c=>(ModelDoc2)c.GetModelDoc2()).ToArray();
  var plans=new[]{solution.Plan.A,solution.Plan.B};var transforms=new[]{solution.TransformA,solution.TransformB};
  Assert(report.Cuts.Count==2&&plans.All(p=>p.Operations.Count==1&&p.Operations[0].Tools.Count==1),"This probe qualifies two existing native folded-butt Blind Cuts only");
  int activateError=0;app.ActivateDoc3(model.GetTitle(),false,(int)swRebuildOnActivation_e.swDontRebuildActiveDoc,ref activateError);
  Assert(app.IsSame(app.ActiveDoc,model)==1&&app.IsSame(assy.GetEditTarget(),model)==1,"Copy Assembly must be active and outside component edit before recording");
  var beforeAssembly=FullTree(model);var beforeParts=parts.Select(FullTree).ToArray();var beforeDefinitions=CutStates(parts,report);
  Assert(beforeDefinitions.Cast<Dictionary<string,object>>().All(x=>(int)x["end_condition"]==(int)swEndConditions_e.swEndCondBlind),"End-reference group must begin with two Blind Cuts");
  row["before_assembly_tree"]=beforeAssembly;row["before_part_trees"]=beforeParts;row["before_cut_definitions"]=beforeDefinitions;
  row["before_in_context_holder_ids"]=HolderIds(beforeAssembly);row["before_assembly_stamp"]=model.GetUpdateStamp();row["before_part_stamps"]=parts.Select(x=>x.GetUpdateStamp()).ToArray();
  var records=new List<object>();bool recording=false,finished=false;Exception linkFailure=null;string group="AFJ_END_REFERENCE_"+Guid.NewGuid().ToString("N");row["undo_group"]=group;
  try{
   Trace.WriteLine("[AFJ-END-UNDO] GROUP_BEGIN "+group);model.Extension.StartRecordingUndoObject();recording=true;
   try{
    for(int i=0;i<2;i++){
     int error=0;app.ActivateDoc3(model.GetTitle(),false,(int)swRebuildOnActivation_e.swDontRebuildActiveDoc,ref error);
     Assert(app.IsSame(app.ActiveDoc,model)==1,"Copy Assembly not active before end-reference edit");
     bool cover=i==0?solution.CoverA:!solution.CoverA;double gap=cover?0:AfjManufacturingConventions.AssemblyFitGap;
     var face=MateFace(components[1-i],transforms[i],plans[i].Operations.Single().Tools.Single(),gap);
     var cut=OwnedCut(parts[i],report,i);model.ClearSelection2(true);Assert(components[i].Select4(false,null,false),"Cannot select exact copied component edit target");
     int info=0;assy.EditPart2(true,false,ref info);Assert(app.IsSame(assy.GetEditTarget(),parts[i])==1,"Wrong edit target inside recorded group");
     try{records.Add(GeometryEndReferenceAdapter.Apply(app,model,components[i],components[1-i],cut,face,gap));}
     finally{assy.EditAssembly();}
     Assert(app.IsSame(assy.GetEditTarget(),model)==1,"Assembly mode not restored inside group");
    }
   }catch(Exception ex){linkFailure=ex;row["link_error"]=ex.ToString();}
   finally{
    // No new Sketch/Cut or parameter changes are allowed between Start/Finish.
    if(app.IsSame(assy.GetEditTarget(),model)!=1)assy.EditAssembly();
    Assert(app.IsSame(app.ActiveDoc,model)==1&&app.IsSame(assy.GetEditTarget(),model)==1,"Cannot finalize the owned Assembly group outside Assembly mode");
    finished=model.Extension.FinishRecordingUndoObject2(group,false);recording=!finished;row["finish_recording_returned"]=finished;
    if(!finished){finished=model.Extension.FinishRecordingUndoObject2(group,false);recording=!finished;row["finish_recording_retry_returned"]=finished;}
    Trace.WriteLine("[AFJ-END-UNDO] GROUP_FINISH finished="+finished);
   }
   Assert(finished&&!recording,"Owned Assembly Undo recording did not finalize; no Undo attempted");
   row["end_references"]=records;
   int afterAssemblyStamp=model.GetUpdateStamp();int[] afterPartStamps=parts.Select(x=>x.GetUpdateStamp()).ToArray();
   var afterAssembly=FullTree(model);var afterParts=parts.Select(FullTree).ToArray();var afterDefinitions=CutStates(parts,report);
   row["after_link_assembly_tree"]=afterAssembly;row["after_link_part_trees"]=afterParts;row["after_link_cut_definitions"]=afterDefinitions;
   row["after_link_in_context_holder_ids"]=HolderIds(afterAssembly);row["after_link_assembly_stamp"]=afterAssemblyStamp;row["after_link_part_stamps"]=afterPartStamps;
   ValidatePair(app,model,a,b,solution,"after_links_before_undo",row);
   // Only read-only observation has occurred since the owned group finished.
   // Require exact identity/stamps/tree before the ONE deliberate Undo call.
   Assert(app.IsSame(app.ActiveDoc,model)==1&&app.IsSame(assy.GetEditTarget(),model)==1,"Assembly context changed before owned Undo");
   Assert(model.GetUpdateStamp()==afterAssemblyStamp&&parts.Select((p,i)=>p.GetUpdateStamp()==afterPartStamps[i]).All(x=>x),"Document stamps changed outside the owned end-reference group; refusing Undo");
   Assert(CanonicalTree(FullTree(model))==CanonicalTree(afterAssembly)&&parts.Select((p,i)=>CanonicalTree(FullTree(p))==CanonicalTree(afterParts[i])).All(x=>x),"Feature history changed outside the owned group; refusing Undo");
   Trace.WriteLine("[AFJ-END-UNDO] EDIT_UNDO_BEGIN count=1");row["undo_calls"]=1;model.EditUndo2(1);Trace.WriteLine("[AFJ-END-UNDO] EDIT_UNDO_END");
   row["after_undo_edit_target_equals_assembly"]=app.IsSame(assy.GetEditTarget(),model)==1;
   var undoAssembly=FullTree(model);var undoParts=parts.Select(FullTree).ToArray();row["after_undo_assembly_tree"]=undoAssembly;row["after_undo_part_trees"]=undoParts;row["after_undo_in_context_holder_ids"]=HolderIds(undoAssembly);
   row["assembly_full_tree_restored"]=CanonicalTree(undoAssembly)==CanonicalTree(beforeAssembly);
   row["part_full_trees_restored"]=parts.Select((p,i)=>CanonicalTree(undoParts[i])==CanonicalTree(beforeParts[i])).All(x=>x);
   row["in_context_holder_ids_restored"]=HolderIds(undoAssembly).SequenceEqual(HolderIds(beforeAssembly));
   var undoneDefinitions=CutStates(parts,report);row["after_undo_cut_definitions"]=undoneDefinitions;
   row["two_blind_definitions_restored"]=json.Serialize(undoneDefinitions)==json.Serialize(beforeDefinitions);
   ValidatePair(app,model,a,b,solution,"after_owned_undo",row);
   bool verified=(bool)row["after_undo_edit_target_equals_assembly"]&&(bool)row["assembly_full_tree_restored"]&&(bool)row["part_full_trees_restored"]&&(bool)row["in_context_holder_ids_restored"]&&(bool)row["two_blind_definitions_restored"];
   row["owned_group_undo_verified"]=verified;
   Assert(verified,"One Assembly-owned end-reference Undo did not restore both exact Blind definitions, geometry, complete Part/Assembly history and original InContext holder IDs");
   if(linkFailure!=null)throw new InvalidOperationException("Link phase failed; Undo was verified but end-reference qualification failed",linkFailure);
   row["status"]="PASS_NATIVE_OWNED_END_REFERENCE_UNDO";
  }catch(Exception ex){row["status"]="FAIL_NATIVE_OWNED_END_REFERENCE_UNDO";row["error"]=ex.ToString();row["hresult"]="0x"+ex.HResult.ToString("X8");throw;}
  finally{row["recording_still_open"]=recording;row["group_finished"]=finished;File.WriteAllText(evidencePath,json.Serialize(row));}
 }

 [STAThread]static int Main(string[] args)
 {
  string root=Path.GetFullPath(args[0]);Directory.CreateDirectory(root);
  Trace.Listeners.Add(new TextWriterTraceListener(Path.Combine(root,"native_apply.private.log")));Trace.AutoFlush=true;
  var rows=new List<object>();
  string[] cases=args.Length>4?args[4].Split(','):new[]{"positive","rollback_after_both_parts","stale_preview_rejected"};
  Assert(cases.All(v=>new[]{"positive","rollback_after_both_parts","stale_preview_rejected","public_preview_then_apply","unsupported_current","shared_part_rejected"}.Contains(v)),"Unknown native case");
  ISldWorks app=null;bool ownsSession=false;
  ModelDoc2 originalActive=null;var originalDocs=new List<ModelDoc2>();
  var originalStamps=new List<int>();var originalDirty=new List<bool>();
  try
  {
   if(System.Environment.GetEnvironmentVariable("AFJ_CREATE_TEST_SW")=="1")
   {
    int[] prior=Process.GetProcessesByName("SLDWORKS").Select(p=>p.Id).OrderBy(p=>p).ToArray();
    Assert(prior.Length<=3,"Too many existing SOLIDWORKS processes; refusing additional activation");
    string allowed=System.Environment.GetEnvironmentVariable("AFJ_EXPECTED_EXISTING_SW_PIDS")??"";
    Assert(String.Join(",",prior)==allowed,"SOLIDWORKS process inventory changed; refusing additional activation");
    Console.WriteLine("AUTHORIZED_OWNED_SESSION_ACTIVATION; apartment="+System.Threading.Thread.CurrentThread.GetApartmentState());
    app=(ISldWorks)Activator.CreateInstance(Type.GetTypeFromProgID("SldWorks.Application.32",true));
    ownsSession=!prior.Contains(app.GetProcessID());
    app.Visible=true;app.UserControl=true;
    Console.WriteLine("DIRECT_COM_SESSION pid="+app.GetProcessID()+"; newlyCreated="+ownsSession+"; revision="+app.RevisionNumber()+"; documents="+app.GetDocumentCount());
   }
   else app=HarnessConnection.AttachVisible();
   if(System.Environment.GetEnvironmentVariable("AFJ_REQUIRE_BLANK_SESSION")=="1")Assert(app.GetDocumentCount()==0,"Isolated test session must be blank before testing");
   int commandId;string pmTitle;bool commandUi;
   app.GetRunningCommandInfo(out commandId,out pmTitle,out commandUi);
   Assert(!commandUi&&!app.CommandInProgress,"An active native command prevents isolated copy testing");
   originalActive=app.ActiveDoc as ModelDoc2;
   for(ModelDoc2 d=app.GetFirstDocument() as ModelDoc2;d!=null;d=d.GetNext() as ModelDoc2)
   {originalDocs.Add(d);originalStamps.Add(d.GetUpdateStamp());originalDirty.Add(d.GetSaveFlag());}
   Console.WriteLine("PRESERVING_EXISTING_DOCUMENTS count="+originalDocs.Count);
   string a=args.Length>2?args[2]:"-fr04-1",b=args.Length>3?args[3]:"-fr03-1";
   foreach(string name in cases)
   {
    var row=(Dictionary<string,object>)Case(app,args[1],root,name,a,b);rows.Add(row);
    if(!((string)row["status"]).StartsWith("PASS_NATIVE",StringComparison.Ordinal))break;
   }
   GC.KeepAlive(app);
  }
  catch(Exception ex){rows.Add(D("status","API_UNAVAILABLE_OR_SETUP_FAILED","error",ex.ToString(),"hresult","0x"+ex.HResult.ToString("X8")));}
  finally
  {
   if(app!=null&&originalDocs.Count>0)try
   {
    if(originalActive!=null){int activateError=0;var restored=app.ActivateDoc3(originalActive.GetTitle(),false,(int)swRebuildOnActivation_e.swDontRebuildActiveDoc,ref activateError);Assert((activateError&~(int)swActivateDocError_e.swDocNeedsRebuildWarning)==0&&app.IsSame(restored,originalActive)==(int)swObjectEquality.swObjectSame,"Could not restore previous active document");}
    bool preserved=originalDocs.Select((d,i)=>app.IsSame(d,app.GetOpenDocumentByName(d.GetPathName()))==1&&d.GetUpdateStamp()==originalStamps[i]&&d.GetSaveFlag()==originalDirty[i]).All(v=>v);
    foreach(var value in rows.Cast<Dictionary<string,object>>())value["preexisting_documents_preserved"]=preserved;
    if(!preserved)rows.Add(D("status","PREEXISTING_STATE_CHANGED_UNVERIFIED"));
   }catch(Exception ex){rows.Add(D("status","PREEXISTING_STATE_CHECK_FAILED","error",ex.ToString()));}
   if(ownsSession&&app!=null)try{if(app.GetDocumentCount()==0){Console.WriteLine("CLOSING_OWNED_EMPTY_SESSION pid="+app.GetProcessID());app.ExitApp();}}catch(Exception ex){Console.WriteLine("OWNED_SESSION_CLEANUP "+ex);}
   File.WriteAllText(Path.Combine(root,"native_apply_results.private.json"),json.Serialize(rows));
   GC.KeepAlive(app);
  }
  Console.WriteLine(json.Serialize(rows));
  return rows.Count==cases.Length&&rows.Cast<Dictionary<string,object>>().All(r=>((string)r["status"]).StartsWith("PASS_NATIVE",StringComparison.Ordinal))?0:1;
 }
