  snap["document_title"]=N(phase+".GetTitle",()=>part.GetTitle());snap["document_path"]=N(phase+".GetPathName",()=>part.GetPathName());snap["update_stamp"]=N(phase+".GetUpdateStamp",()=>part.GetUpdateStamp());snap["needs_save_in_memory"]=N(phase+".GetSaveFlag",()=>part.GetSaveFlag());
  var seen=new HashSet<int>();int count=0;Action<Feature,bool,string,int> visit=null;
  visit=(first,top,parent,depth)=>{

   for(Feature f=first;f!=null;){
    Require(++count<=1600,"Feature count bound exceeded");string p=phase+".feature["+depth+","+(i++)+"]";int id=N(p+".GetID",()=>f.GetID());Require(siblings.Add(id),"Repeated sibling Feature ID");
    if(seen.Add(id)){
     string type=N(p+".GetTypeName2",()=>f.GetTypeName2()),underlying=type=="ICE"?N(p+".GetTypeName",()=>f.GetTypeName()):type;

  if(reviewed!=null){snap["symmetric_difference_from_reviewed_m3"]=N(phase+".CurrentBodyCopy.SymmetricDifference",()=>AfjNativeCutCommit.SymmetricDifference(AfjTemporaryBoolean.Copy(AfjNativeCutCommit.CurrentBody(part)),reviewed));snap["reviewed_geometry_matches"]=(double)snap["symmetric_difference_from_reviewed_m3"]<=1e-12;}
