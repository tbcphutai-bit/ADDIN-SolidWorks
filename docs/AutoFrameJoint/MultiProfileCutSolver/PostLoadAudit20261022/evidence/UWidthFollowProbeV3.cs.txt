// V3 measures BOTH actual endpoints: a source width dimension may hold the far side fixed.
// Historical V1/V2 fixed-endpoint failures remain separate incomplete probes.
// Detached, copied-CAD diagnostic. No solver rerun, Cut, Save or main build.
using System;
using System.Linq;
using System.Collections.Generic;
using SolidWorks.Interop.sldworks;
using SolidWorks.Interop.swconst;
using ADDIN.Commands.AutoFrameJoint;
internal static class UWidthFollowProbeV3
{
 public sealed class Gauge { public string Phase;public double Width,SourceDimension,NormalGap,ParallelCosine,SignedGap;public int MasterStatus,LoserStatus,MasterError,LoserError;public double[] MasterFarNative,LoserFarNative,MasterTransform,LoserTransform;public double[][] MasterEndpointsNative,LoserEndpointsNative;public int MasterDangling,MasterOver,LoserDangling,LoserOver; }
 public sealed class Result { public string Status="UNVERIFIED",Error,Scope="COPY_NATIVE_WEB_WIDTH_CHANGE_SKETCH_FOLLOW_ONLY_NO_CUT";public bool SolverRerun=false,SaveCalled=false,Restored=false,AssemblyContextRestored=false;public int SourceSketchId,ChangeSetStatus=-1,RestoreSetStatus=-1;public string SourceDimension;public double BeforeWidth,RequestedWidth,MeasuredMasterFarNativeMovement,RestoredMasterFarNativeDifference,MeasuredLargestMasterEndpointMovement,RestoredLargestMasterEndpointDifference,RestoredLargestLoserEndpointDifference,RestoredTransformMaximumDifference;public readonly List<Gauge> Gauges=new List<Gauge>(); }
 static void Need(bool b,string why){if(!b)throw new InvalidOperationException("WIDTH_FOLLOW_UNVERIFIED: "+why);}
 static double[] Sub(double[] a,double[] b){return a.Zip(b,(x,y)=>x-y).ToArray();}static double Dot(double[] a,double[] b){return a.Zip(b,(x,y)=>x*y).Sum();}static double Norm(double[] a){return Math.Sqrt(Dot(a,a));}static double[] Unit(double[] a){double n=Norm(a);Need(n>1e-9,"zero vector");return a.Select(x=>x/n).ToArray();}
 static double[] Cross(double[] a,double[] b){return new[]{a[1]*b[2]-a[2]*b[1],a[2]*b[0]-a[0]*b[2],a[0]*b[1]-a[1]*b[0]};}
 static double[] T(double[] p,MathTransform transform){return AfjProfileAnalyzer.Transform(p,transform.ArrayData as double[]);}
 static Feature Find(ModelDoc2 part,int id){int n=0;for(Feature f=part.FirstFeature() as Feature;f!=null;f=f.GetNextFeature() as Feature){Need(++n<2000,"feature traversal bound");if(f.GetID()==id)return f;}return null;}
 static string Id(SketchSegment s){return String.Join(",",((Array)s.GetID()).Cast<object>().Select(Convert.ToString));}
 static double[][] NativeEnds(Sketch sketch,SketchSegment segment){var line=segment as SketchLine;Need(line!=null,"straight seam required");var a=line.GetStartPoint2() as SketchPoint;var b=line.GetEndPoint2() as SketchPoint;var inv=sketch.ModelToSketchTransform.Inverse() as MathTransform;return new[]{T(new[]{a.X,a.Y,a.Z},inv),T(new[]{b.X,b.Y,b.Z},inv)};}
 static Sketch Owned(ModelDoc2 part,int id){var f=Find(part,id);Need(f!=null,"owned Sketch missing");var sketch=f.GetSpecificFeature2() as Sketch;Need(sketch!=null&&!sketch.Is3D(),"native 2D Sketch required");return sketch;}
 static SketchSegment Seam(Sketch sketch,string id){var a=sketch.GetSketchSegments() as Array;Need(a!=null,"Sketch inventory unavailable");return a.Cast<object>().OfType<SketchSegment>().Single(s=>Id(s)==id&&!s.ConstructionGeometry);}
 static string FindDiagonal(ModelDoc2 part,Component2 component,int id,double[] u,double[] v){var sketch=Owned(part,id);var a=sketch.GetSketchSegments() as Array;var hits=new List<string>();foreach(var segment in a.Cast<object>().OfType<SketchSegment>()){if(segment.ConstructionGeometry)continue;var ends=NativeEnds(sketch,segment);var world=ends.Select(p=>T(p,component.GetTotalTransform(false))).ToArray();var d=Unit(Sub(world[1],world[0]));if(Math.Abs(Dot(d,u))>1e-5&&Math.Abs(Dot(d,v))>1e-5)hits.Add(Id(segment));}Need(hits.Count==1,"one actual nonorthogonal seam required");return hits[0];}
 static Dimension ReadDimension(ModelDoc2 part,int id,string fullName){var f=Find(part,id);Need(f!=null,"source original Sketch missing");for(DisplayDimension d=f.GetFirstDisplayDimension() as DisplayDimension;d!=null;d=f.GetNextDisplayDimension(d) as DisplayDimension){var dim=d.GetDimension2(0) as Dimension;if(dim!=null&&dim.FullName==fullName)return dim;}throw new InvalidOperationException("Source native dimension identity missing");}
 static Body2 Body(Component2 component){object info;var bodies=component.GetBodies3((int)swBodyType_e.swSolidBody,out info) as Array;Need(bodies!=null&&bodies.Length==1,"one actual native solid required");return bodies.GetValue(0) as Body2;}
 static double Span(Body2 body,double[] axis){Need(body.Check2()==0,"native body check failed");double x,y,z;Need(body.GetExtremePoint(axis[0],axis[1],axis[2],out x,out y,out z),"positive B-rep support failed");double hi=Dot(new[]{x,y,z},axis);Need(body.GetExtremePoint(-axis[0],-axis[1],-axis[2],out x,out y,out z),"negative B-rep support failed");return hi-Dot(new[]{x,y,z},axis);}
 static void Activate(ISldWorks app,ModelDoc2 model){int error=0;var returned=app.ActivateDoc3(model.GetTitle(),false,(int)swRebuildOnActivation_e.swDontRebuildActiveDoc,ref error) as ModelDoc2;Need(returned!=null&&app.IsSame(returned,model)==(int)swObjectEquality.swObjectSame&&(error&~(int)swActivateDocError_e.swDocNeedsRebuildWarning)==0,"exact copy activation failed");}
 static void Rebuild(ISldWorks app,ModelDoc2 model,ModelDoc2 part){Activate(app,part);Need(part.EditRebuild3(),"copied source Part rebuild returned false");Activate(app,model);Need(model.EditRebuild3(),"copied Assembly rebuild returned false");}
 public static Result Run(ISldWorks app,ModelDoc2 assembly,Component2 longFrame,Component2 shortFrame,int masterId,int loserId,double[] u,double[] v,double expectedWidth,double expectedGap,Action<string> stage)
 {
  var r=new Result();ModelDoc2 part=shortFrame.GetModelDoc2() as ModelDoc2,longPart=longFrame.GetModelDoc2() as ModelDoc2;bool changed=false;double old=0;string name=null;
  Action<string> mark=s=>{if(stage!=null)stage(s);};
  try{
   Need(part!=null&&longPart!=null,"resolved copied Parts required");mark("IDENTIFY_ACTUAL_SOURCE_WEB_WIDTH_PARAMETER");
   var flange=new List<Feature>();for(Feature f=part.FirstFeature() as Feature;f!=null;f=f.GetNextFeature() as Feature)if(f.GetTypeName2()=="SMBaseFlange")flange.Add(f);Need(flange.Count==1,"one source base flange required");
   var parents=flange[0].GetParents() as Array;var sourceSketch=parents.Cast<object>().OfType<Feature>().Where(f=>f.GetSpecificFeature2() is Sketch).ToArray();Need(sourceSketch.Length==1,"one original profile Sketch required");r.SourceSketchId=sourceSketch[0].GetID();
   var dims=new List<Dimension>();for(DisplayDimension d=sourceSketch[0].GetFirstDisplayDimension() as DisplayDimension;d!=null;d=sourceSketch[0].GetNextDisplayDimension(d) as DisplayDimension){var dim=d.GetDimension2(0) as Dimension;if(dim!=null&&dim.DrivenState==(int)swDimensionDrivenState_e.swDimensionDriving&&Math.Abs(dim.SystemValue-expectedWidth)<1e-8)dims.Add(dim);}Need(dims.Count==1,"one source driving dimension matching measured Web width required");
   name=dims[0].FullName;r.SourceDimension=name;old=dims[0].SystemValue;r.BeforeWidth=old;r.RequestedWidth=old+.001;
   var inv=shortFrame.GetTotalTransform(false).Inverse() as MathTransform;var nativeAxis=Unit(Sub(T(u,inv),T(new double[3],inv)));Need(Math.Abs(Span(Body(shortFrame),nativeAxis)-old)<1e-8,"source dimension not confirmed by native B-rep width span");
   string masterSeam=FindDiagonal(part,shortFrame,masterId,u,v),loserSeam=FindDiagonal(longPart,longFrame,loserId,u,v);
   double? expectedSignedGap=null;Func<string,Gauge> gauge=phase=>{
    var ms=Owned(part,masterId);var ls=Owned(longPart,loserId);var mn=NativeEnds(ms,Seam(ms,masterSeam));var ln=NativeEnds(ls,Seam(ls,loserSeam));var mt=shortFrame.GetTotalTransform(false);var lt=longFrame.GetTotalTransform(false);var mw=mn.Select(p=>T(p,mt)).ToArray();var lw=ln.Select(p=>T(p,lt)).ToArray();var md=Unit(Sub(mw[1],mw[0]));var ld=Unit(Sub(lw[1],lw[0]));bool warn;int me=Find(part,masterId).GetErrorCode2(out warn),le=Find(longPart,loserId).GetErrorCode2(out warn);
    var g=new Gauge{Phase=phase,Width=Span(Body(shortFrame),nativeAxis),SourceDimension=ReadDimension(part,r.SourceSketchId,name).SystemValue,NormalGap=Norm(Cross(Sub(lw[0],mw[0]),md)),SignedGap=Dot(Cross(Sub(lw[0],mw[0]),md),Unit(Cross(u,v))),ParallelCosine=Math.Abs(Dot(md,ld)),MasterStatus=ms.GetConstrainedStatus(),LoserStatus=ls.GetConstrainedStatus(),MasterError=me,LoserError=le,MasterFarNative=mn[Dot(mw[1],u)>Dot(mw[0],u)?1:0],LoserFarNative=ln[Dot(lw[1],v)>Dot(lw[0],v)?1:0],MasterTransform=mt.ArrayData as double[],LoserTransform=lt.ArrayData as double[],MasterEndpointsNative=mn,LoserEndpointsNative=ln,MasterDangling=ms.RelationManager.GetRelationsCount((int)swSketchRelationFilterType_e.swDangling),MasterOver=ms.RelationManager.GetRelationsCount((int)swSketchRelationFilterType_e.swOverDefining),LoserDangling=ls.RelationManager.GetRelationsCount((int)swSketchRelationFilterType_e.swDangling),LoserOver=ls.RelationManager.GetRelationsCount((int)swSketchRelationFilterType_e.swOverDefining)};
    r.Gauges.Add(g);if(!expectedSignedGap.HasValue)expectedSignedGap=Math.Sign(g.SignedGap)*expectedGap;Need(Math.Abs(g.SignedGap-expectedSignedGap.Value)<1e-8,"signed assembly gap changed side or magnitude");Need(g.MasterStatus==3&&g.LoserStatus==3&&g.MasterDangling==0&&g.MasterOver==0&&g.LoserDangling==0&&g.LoserOver==0&&me==0&&le==0&&Math.Abs(g.ParallelCosine-1)<1e-8&&Math.Abs(g.NormalGap-expectedGap)<1e-8,"native constraint/parallel/normal-gap readback failed");return g;};
   mark("MEASURE_BEFORE");var before=gauge("BEFORE_WIDTH_CHANGE");
   mark("SET_SOURCE_WIDTH_PLUS_1MM_COPY_ONLY");Activate(app,part);changed=true;r.ChangeSetStatus=ReadDimension(part,r.SourceSketchId,name).SetSystemValue3(r.RequestedWidth,(int)swSetValueInConfiguration_e.swSetValue_InThisConfiguration,null);Need(r.ChangeSetStatus==(int)swSetValueReturnStatus_e.swSetValue_Successful,"native source parameter change failed");Rebuild(app,assembly,part);
   mark("MEASURE_NATIVE_FOLLOW_WITHOUT_SOLVER_RERUN");var after=gauge("AFTER_PLUS_1MM");Need(Math.Abs(after.Width-r.RequestedWidth)<1e-8&&Math.Abs(after.SourceDimension-r.RequestedWidth)<1e-8,"native B-rep width did not follow source dimension");r.MeasuredMasterFarNativeMovement=Norm(Sub(after.MasterFarNative,before.MasterFarNative));r.MeasuredLargestMasterEndpointMovement=after.MasterEndpointsNative.Max((double[] p)=>before.MasterEndpointsNative.Min((double[] b)=>Norm(Sub(p,b))));Need(r.MeasuredLargestMasterEndpointMovement>1e-7,"neither actual master endpoint moved after native source width change");
   mark("RESTORE_SOURCE_WIDTH_COPY_ONLY");Activate(app,part);r.RestoreSetStatus=ReadDimension(part,r.SourceSketchId,name).SetSystemValue3(old,(int)swSetValueInConfiguration_e.swSetValue_InThisConfiguration,null);Need(r.RestoreSetStatus==(int)swSetValueReturnStatus_e.swSetValue_Successful,"native source restoration failed");Rebuild(app,assembly,part);changed=false;
   var restored=gauge("RESTORED_ORIGINAL_WIDTH");r.RestoredMasterFarNativeDifference=Norm(Sub(restored.MasterFarNative,before.MasterFarNative));r.RestoredLargestMasterEndpointDifference=restored.MasterEndpointsNative.Max((double[] p)=>before.MasterEndpointsNative.Min((double[] b)=>Norm(Sub(p,b))));r.RestoredLargestLoserEndpointDifference=restored.LoserEndpointsNative.Max((double[] p)=>before.LoserEndpointsNative.Min((double[] b)=>Norm(Sub(p,b))));r.RestoredTransformMaximumDifference=restored.MasterTransform.Zip(before.MasterTransform,(a,b)=>Math.Abs(a-b)).Concat(restored.LoserTransform.Zip(before.LoserTransform,(a,b)=>Math.Abs(a-b))).Max();r.Restored=Math.Abs(restored.Width-old)<1e-8&&r.RestoredLargestMasterEndpointDifference<1e-8&&r.RestoredLargestLoserEndpointDifference<1e-8&&r.RestoredTransformMaximumDifference<1e-10;Need(r.Restored,"native width/seam restoration failed");
   r.AssemblyContextRestored=app.IsSame(((AssemblyDoc)assembly).GetEditTarget(),assembly)==(int)swObjectEquality.swObjectSame;Need(r.AssemblyContextRestored,"assembly edit context not restored");r.Status="PASS_COPY_NATIVE_WIDTH_FOLLOW_SKETCHES_ONLY_NO_CUT_NO_SOLVER_RERUN";
  }catch(Exception ex){r.Status="FAIL_OR_UNVERIFIED";r.Error="HRESULT=0x"+ex.HResult.ToString("X8")+" "+ex;}
  finally{if(changed&&name!=null)try{Activate(app,part);r.RestoreSetStatus=ReadDimension(part,r.SourceSketchId,name).SetSystemValue3(old,(int)swSetValueInConfiguration_e.swSetValue_InThisConfiguration,null);Rebuild(app,assembly,part);r.Restored=Math.Abs(ReadDimension(part,r.SourceSketchId,name).SystemValue-old)<1e-8;}catch(Exception ex){r.Error+=" RESTORE_FAILED "+ex;r.Restored=false;}}
  try{r.AssemblyContextRestored=app.IsSame(app.ActiveDoc,assembly)==(int)swObjectEquality.swObjectSame&&app.IsSame(((AssemblyDoc)assembly).GetEditTarget(),assembly)==(int)swObjectEquality.swObjectSame;}catch{}
  return r;
 }
