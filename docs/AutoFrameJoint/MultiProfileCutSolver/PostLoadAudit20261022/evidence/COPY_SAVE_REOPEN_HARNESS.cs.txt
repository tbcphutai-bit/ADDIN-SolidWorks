  visit=(first,top,parent,depth)=>{

    Require(++count<=1600,"Feature count bound exceeded");string p=phase+".feature["+depth+","+(i++)+"]";int id=N(p+".GetID",()=>f.GetID());Require(siblings.Add(id),"Repeated sibling Feature ID");

  var all=rows.Cast<Dictionary<string,object>>().ToArray();var added=all.Where(r=>(bool)r["new_since_baseline"]).ToArray();var sketches=added.Where(r=>r.ContainsKey("sketch")).ToArray();

 static void CloseThenReopenReadOnly(ISldWorks app,string folder,List<object> descriptors){

   string frame=(string)desc["frame"],path=(string)desc["copy_path"],p=frame+".ROUNDTRIP.REOPEN_READONLY";
   Require(Inside(path,folder),"OpenDoc6 path escaped disposable directory");

 }
