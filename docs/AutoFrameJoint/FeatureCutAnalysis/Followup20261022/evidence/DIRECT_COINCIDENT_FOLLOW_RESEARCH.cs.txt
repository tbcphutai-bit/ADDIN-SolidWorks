 class Pair {internal Face2 Own,Other;internal Edge Edge;internal Vertex A,B;internal double Area;internal double[] WorldA,WorldB;}
 static Pair References(ISldWorks sw,Component2 own,Component2 other)
 {
  var ot=own.GetTotalTransform(false);var rt=other.GetTotalTransform(false);
  var op=(ModelDoc2)own.GetModelDoc2();var rp=(ModelDoc2)other.GetModelDoc2();
  double[] oa=Unit(Vector(sw,Axis(op),ot)),ra=Unit(Vector(sw,Axis(rp),rt));
  Require(Math.Abs(Dot(oa,ra))<1e-7,"Probe requires perpendicular extrusion axes");double[] depth=Unit(Cross(oa,ra));
  var pairs=new List<Pair>();
  foreach(Face2 of in (Array)Body(own).GetFaces()){
   var os=(Surface)of.GetSurface();if(!os.IsPlane())continue;double[] po=(double[])os.PlaneParams;
   var no=Unit(Vector(sw,po.Take(3).ToArray(),ot));if(Math.Abs(Dot(no,depth))<.999999)continue;
   var wo=Point(sw,po.Skip(3).Take(3).ToArray(),ot);
   foreach(Face2 rf in (Array)Body(other).GetFaces()){
    var rs=(Surface)rf.GetSurface();if(!rs.IsPlane())continue;double[] pr=(double[])rs.PlaneParams;
    var nr=Unit(Vector(sw,pr.Take(3).ToArray(),rt));if(Math.Abs(Dot(nr,depth))<.999999)continue;
    var wr=Point(sw,pr.Skip(3).Take(3).ToArray(),rt);if(Math.Abs(Dot(Sub(wr,wo),depth))>1e-8)continue;

 static PointGuide PointReferences(ISldWorks sw,Component2 own,Component2 other,Face2 ownFace,Sketch sketch,bool projection,bool sketchInputWorld)
 {
  MathTransform ot=own.GetTotalTransform(false),rt=other.GetTotalTransform(false),inv=(MathTransform)ot.Inverse();
  double[] axis=Unit(Vector(sw,Axis((ModelDoc2)own.GetModelDoc2()),ot));
  if(projection){
   foreach(Face2 f in (Array)Body(other).GetFaces())foreach(Edge e in (Array)f.GetEdges())foreach(Vertex vertex in new[]{e.GetStartVertex() as Vertex,e.GetEndVertex() as Vertex}){

   }
   throw new InvalidOperationException("No noncoplanar actual other-Frame Vertex found");
  }
  foreach(Face2 f in (Array)Body(other).GetFaces()){
   var surface=(Surface)f.GetSurface();if(!surface.IsPlane())continue;double[] p=(double[])surface.PlaneParams;
   double[] n=Unit(Vector(sw,p.Take(3).ToArray(),rt)),plane=Point(sw,p.Skip(3).Take(3).ToArray(),rt);if(Math.Abs(Dot(n,axis))<.999999)continue;
   foreach(Edge e in (Array)ownFace.GetEdges()){
    var cv=e.GetCurve() as Curve;if(cv==null||!cv.IsLine())continue;var va=e.GetStartVertex() as Vertex;var vb=e.GetEndVertex() as Vertex;if(va==null||vb==null)continue;
    double[] a=Point(sw,(double[])va.GetPoint(),ot),b=Point(sw,(double[])vb.GetPoint(),ot),d=Sub(b,a);if(Math.Sqrt(Dot(d,d))<.02||Math.Abs(Dot(Unit(d),axis))<.999999)continue;

   }
