_HELPER=Path(__file__).with_name("WebFaceOffsetWriter.cs.txt").read_text(encoding="utf-8")
_MARK="AFJ-U-WEB-FACE-OFFSET-PROBE-20261022"

def _once(source,old,new):
    if source.count(old)!=1:
        raise ValueError("Expected one source anchor: "+old[:100])
    return source.replace(old,new,1)

    end="        private static void ApplyPart(ISldWorks app, PartEdit edit, AfjCutPlan plan,"
